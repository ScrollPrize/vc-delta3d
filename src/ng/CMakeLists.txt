cmake_minimum_required(VERSION 3.22)
project(volcodec_ng_hybrid LANGUAGES C CXX)
set(VC_DELTA3D_SOURCE "" CACHE PATH "Existing vc-delta3d v0.2.0 source checkout (no downloads)")
if(NOT EXISTS "${VC_DELTA3D_SOURCE}/include/vc_delta3d/codec.hpp")
    message(FATAL_ERROR "Set VC_DELTA3D_SOURCE to an existing vc-delta3d v0.2.0 checkout")
endif()
set(VC_DELTA3D_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(VC_DELTA3D_INSTALL_CPP OFF CACHE BOOL "" FORCE)
set(VC_DELTA3D_BUILD_NG OFF CACHE BOOL "" FORCE)
add_subdirectory("${VC_DELTA3D_SOURCE}" delta3d)
# Mechanical adaptation of the one shared upstream implementation. A TLS array
# makes GCC's PIC build call __tls_get_addr inside the per-slot/per-symbol loops.
# This 16KiB table is rebuilt completely on every decode, so stack storage is
# equivalent and avoids both TLS accesses and process-wide shared mutable state.
# Keep the external checkout untouched; fail if its expected declaration drifts.
file(READ "${VC_DELTA3D_SOURCE}/src/codec.cpp" DELTA_ADAPTED_SOURCE)
set(DELTA_TABLE_DECL "static thread_local std::uint32_t slotEntry[kRansM];")
string(FIND "${DELTA_ADAPTED_SOURCE}" "${DELTA_TABLE_DECL}" DELTA_TABLE_OFFSET)
# Integrated Delta3D already contains this optimization; replacement is a
# no-op there. Keep compatibility with the pinned original checkout too.
string(REPLACE "${DELTA_TABLE_DECL}" "std::uint32_t slotEntry[kRansM];" DELTA_ADAPTED_SOURCE "${DELTA_ADAPTED_SOURCE}")
# A one-symbol rANS alphabet leaves all eight states unchanged and emits zero
# renormalization words. Keep the same normalization/table/state writer, but
# omit the redundant symbol scan and oversized buffer in that exact case.
set(DELTA_STREAM_CAP "const std::size_t streamCap = 2 * n + kRansStateBytes + 64;")
set(DELTA_ENCODE_BEGIN "std::size_t j = n;")
string(REPLACE "${DELTA_STREAM_CAP}" "const bool constant = std::find(freq, freq + 256, kRansM) != freq + 256;\n    const std::size_t streamCap = (constant ? 0 : 2 * n) + kRansStateBytes + 64;" DELTA_ADAPTED_SOURCE "${DELTA_ADAPTED_SOURCE}")
string(REPLACE "${DELTA_ENCODE_BEGIN}" "std::size_t j = constant ? 0 : n;" DELTA_ADAPTED_SOURCE "${DELTA_ADAPTED_SOURCE}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/delta_adapter")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/delta_adapter/codec.cpp" "${DELTA_ADAPTED_SOURCE}")
set_property(TARGET vc_delta3d PROPERTY SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/delta_bridge.cpp")
target_include_directories(vc_delta3d BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/delta_adapter" "${VC_DELTA3D_SOURCE}/src")
find_package(Threads REQUIRED)
find_path(NG_ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(NG_ZSTD_LIBRARY NAMES zstd REQUIRED)
set(NG_FIXED_OBJECTS)
foreach(dim IN ITEMS 32 64 128 256)
    add_library(ng_fixed${dim} OBJECT ng_fixed.c)
    set_property(TARGET ng_fixed${dim} PROPERTY POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(ng_fixed${dim} PRIVATE NG_FIXED_GEOMETRY=${dim} NG_BLOCK_DIM=${dim})
    if(NOT MSVC)
        target_compile_options(ng_fixed${dim} PRIVATE -std=gnu2x -ffp-contract=off)
    else()
        target_compile_options(ng_fixed${dim} PRIVATE /std:clatest)
    endif()
    list(APPEND NG_FIXED_OBJECTS $<TARGET_OBJECTS:ng_fixed${dim}>)
endforeach()
add_library(volcodec_ng_hybrid SHARED volcodec_ng_lib.c ${NG_FIXED_OBJECTS} hybrid.cpp cabac.cpp ng_delta_experiment.cpp ng_axis_experiment.cpp joint_subchunks.cpp)
target_include_directories(volcodec_ng_hybrid PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(volcodec_ng_hybrid PUBLIC vc_delta3d Threads::Threads)
target_include_directories(volcodec_ng_hybrid PRIVATE ${NG_ZSTD_INCLUDE_DIR})
target_link_libraries(volcodec_ng_hybrid PRIVATE ${NG_ZSTD_LIBRARY})
if(NOT MSVC)
    target_compile_options(volcodec_ng_hybrid PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu2x> -ffp-contract=off)
    target_link_libraries(volcodec_ng_hybrid PRIVATE m)
else()
    target_compile_options(volcodec_ng_hybrid PRIVATE $<$<COMPILE_LANGUAGE:C>:/std:clatest>)
endif()
add_library(benchmark_hybrid SHARED ../../scripts/paris4_codec_ladder_native.cpp)
target_compile_definitions(benchmark_hybrid PRIVATE BENCH_HYBRID)
target_link_libraries(benchmark_hybrid PRIVATE volcodec_ng_hybrid)

# Experimental 64-cube build of the same kernels; not a production codec.
add_library(ng_blocks64 SHARED volcodec_ng_lib.c ${NG_FIXED_OBJECTS} ng_delta_experiment.cpp delta_bridge.cpp)
target_compile_features(ng_blocks64 PRIVATE cxx_std_20)
target_compile_definitions(ng_blocks64 PRIVATE NG_BLOCK_DIM=64)
target_include_directories(ng_blocks64 PRIVATE ${NG_ZSTD_INCLUDE_DIR})
target_link_libraries(ng_blocks64 PRIVATE ${NG_ZSTD_LIBRARY})
target_include_directories(ng_blocks64 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/delta_adapter" "${VC_DELTA3D_SOURCE}/src" "${VC_DELTA3D_SOURCE}/include")
if(NOT MSVC)
    target_compile_options(ng_blocks64 PRIVATE $<$<COMPILE_LANGUAGE:C>:-std=gnu2x> -ffp-contract=off)
    target_link_libraries(ng_blocks64 PRIVATE m Threads::Threads)
endif()
